\section{Pythonnet}
Naast de bedoeling om de code te documenteren, is het ook de bedoeling om de codebase die in \texttt{C#} is geschreven te laten interageren met python. De onderzoekers kennen geen \texttt{C#} maar wel python. Omdat zij groten deels beslissen wat de berekeningen zijn kunnen ze dat wel in python zetten.
Via \textcite{pythonnet} kan python dll's lezen, importeren en gebruiken.

    from System import String
    from System.Collections import *

\end{listing}
\begin{listing}
  \begin{minted}{python}
    import clr
    clr.AddReference("System.Windows.Forms")
    clr.AddReference("System.Drawing")
    clr.AddReference("System")
    clr.AddReference("System.Windows")
    from pythonnet import load
    from System import *
    from System.Drawing import Color, Point, Size
    from System.Windows.Forms import Application, Button, Form, MessageBox, TextBox
    from System.Windows import Clipboard, DataFormats, Window
    from System.Windows.Threading import Dispatcher
class MainForm(Form):
    def __init__(self):
        super().__init__()
        self.Text = ".NET in python"
        self.Size = Size(400, 600)
def simple_form():
    global _application_initialized
    
    try:
        if not _application_initialized:
            Application.EnableVisualStyles()
            Application.SetCompatibleTextRenderingDefault(False)
            _application_initialized = True
        
        Application.Run(MainForm())
    except Exception as e:
        print(f"Error in simple_form: {e}")
        return None


  \end{minted}
