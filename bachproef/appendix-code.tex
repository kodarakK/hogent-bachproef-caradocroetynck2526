private static void SetupDll()
{
    var assemblyLocation = Path.GetDirectoryName(
        Assembly.GetExecutingAssembly().Location)!;
    var solutionRoot = Path.GetFullPath(
        Path.Combine(assemblyLocation, "..", "..", "..", "..")
    );

    var targetDir = Path.Combine(solutionRoot, "ILVO.EMAV.Jupyter", "dlls");
    Directory.CreateDirectory(targetDir);

    var dllFiles = Directory.GetFiles(solutionRoot, "*.dll", SearchOption.AllDirectories)
        .Where(path => path.Contains(@"\obj\"))
        .Where(path => !path.Contains(@"\ref\"))
        .Where(path => !path.Contains(@"\runtimes\"));

    foreach (var dll in dllFiles)
    {
        var fileName = Path.GetFileName(dll);
        var destPath = Path.Combine(targetDir, fileName);
        File.Copy(dll, destPath, overwrite: true);
        Console.WriteLine($"Copied: {fileName}");
    }
}
\end{minted}
