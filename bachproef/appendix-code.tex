        factors = await_task(startup.factor_repo.GetAll())

        job_id = startup.Guid.NewGuid()
        user_id = startup.Guid.NewGuid()
        db_path = os.path.join(os.getcwd(), f"calculation.integration.2015.{job_id}.db")

        calc_metadata = startup.CalculationOutputMetadata()
        calc_metadata.CalculationFactorMetadataId = factor.Id
        calc_metadata.InputMetadataId = input_metadata.Id
        calc_metadata.DbFileName = db_path
        calc_metadata.JobMetadataId = job_id
        calc_metadata.UserID = user_id

        await_task(calc_output_repo.Create(calc_metadata))

        # run calculation up to output tab (includes animal subcategory summary)
        emission_service = startup.service_provider.GetService(startup.IEmissionCalculationService)
        payload = startup.EmissionCalcPayload(input_metadata.Id, user_id, factor.Id, job_id)

        print("Starting calculation (this may take a while)...")
        await_task(emission_service.CalculateAll(payload, startup.EmissionCalcStep.OutputTabAfterGIS, False))
        print("Calculation finished.")

        # verify output DB exists
        if not os.path.isfile(db_path):
            raise RuntimeError(f"Expected output DB not found at {db_path}")

        # read output DierSubCategorieInfo
        conn = sqlite3.connect(db_path)
        cur = conn.cursor()

        cur.execute("SELECT Code, NbDieren FROM DierSubCategorieInfo")
        rows = cur.fetchall()
        conn.close()

        if not rows:
            raise AssertionError("No rows found in DierSubCategorieInfo in calculation output DB")

        # get expected subcategory codes from rekenfactor repository
        tbl_repo = startup.service_provider.GetService(startup.ICalculationFactorBaseRepository[startup.TblDierSubCategorie])

    """
Simulated BerekenEmissieVoedertechnieken module called from C# via pythonnet.

Public function:
  Bereken(emissie_stallen, regressie_list=None, regressie_rechte_list=None, convenant_list=None)

Behavior (implements the original C# logic):
 - For each EmissieStal element compute emissie = min(TmpEmissieNH3Gereduceerd, TmpEmissieNH3Trad, TmpEmissieNH3StalEF)
 - If emissie > 0 and AantalDierenPASVoeding == 0 and AantalDierenPASCombi == 0 then:
     - if ea.Regressie: try to find a matching regressieRechte (by TblDierSubCategorie.Code == DierSubCategorieCode)
         compute X = ((ProductieN / AantalDieren) - B) / A (safe-guard A != 0)
         find regression entry where X < Regessiemax (chooses the smallest Regessiemax that is > X)
         if found and not CodeStalType startswith 'V ' or 'P ' and AantalDierenTechniek == 0
             emissie *= (1 - Reductiepercentage/100)
     - elif ea.Convenant: find convenant entry and if ProductieN / AantalDieren <= Convenantmax
             emissie *= (1 - Reductiepercentage/100)
 - Set ea.EmissieNH3 = emissie
 - Return the (possibly modified) emissie_stallen sequence (as a Python list)

Notes:
 - This module is defensive: it accepts CLR objects (pythonnet proxies) or plain dicts.
 - If regressie/convenant lists are not provided, it will try to read them from the passed objects (if they contain links), else it uses no-op fallbacks.
 - Designed for simulation and predictable behavior when called from C# for testing/documentation.
"""

from typing import Iterable, List, Any, Optional


def _get(obj: Any, name: str, default=None):
    """Robust property accessor for CLR proxies and dicts."""
    if obj is None:
        return default
    # dict-like
    try:
        return obj[name]
    except Exception:
        pass
    try:
        return getattr(obj, name)
    except Exception:
        pass
    try:
        # some CLR proxies expose methods
        prop = getattr(obj, name, default)
        return prop
    except Exception:
        return default


def _set(obj: Any, name: str, value):
    try:
        obj[name] = value
        return
    except Exception:
        pass
    try:
        setattr(obj, name, value)
        return
    except Exception:
        pass
    # last resort: ignore


def _to_float(v, default=0.0):
    try:
        if v is None:
            return float(default)
        return float(v)
    except Exception:
        return float(default)


def _starts_with(v: Any, prefix: str) -> bool:
    try:
        if v is None:
            return False
        return str(v).startswith(prefix)
    except Exception:
        return False


# helper to find a single item in a list by nested property: item.<nested>.<prop> == value
def _find_first_by_nested_code(collection: Iterable[Any], nested_attr: str, prop_name: str, value: str):
    if collection is None:
        return None
    for it in collection:
        nested = _get(it, nested_attr)
        if nested is None:
            continue
        code = _get(nested, prop_name)
        if code is None:
            continue
        if str(code).strip() == str(value).strip():
            return it
    return None


def Bereken(emissie_stallen: Iterable[Any],
           regressie_list: Optional[Iterable[Any]] = None,
           regressie_rechte_list: Optional[Iterable[Any]] = None,
           convenant_list: Optional[Iterable[Any]] = None) -> List[Any]:
    """Compute EmissieNH3 for each EmissieStal. Returns list of same objects with EmissieNH3 set.

    Parameters are permissive to accept CLR proxies or plain python dicts.
    """
    out = []

    # normalize collections to lists
    regressie_list = list(regressie_list) if regressie_list is not None else []
    regressie_rechte_list = list(regressie_rechte_list) if regressie_rechte_list is not None else []
    convenant_list = list(convenant_list) if convenant_list is not None else []

    for ea in emissie_stallen:
        # read tmp emissions, fallback to 0
        t1 = _to_float(_get(ea, 'TmpEmissieNH3Gereduceerd', 0.0))
        t2 = _to_float(_get(ea, 'TmpEmissieNH3Trad', 0.0))
        t3 = _to_float(_get(ea, 'TmpEmissieNH3StalEF', 0.0))
        emissie = min(t1, t2, t3)

        aantal_pas_voeding = _to_float(_get(ea, 'AantalDierenPASVoeding', 0))
        aantal_pas_combi = _to_float(_get(ea, 'AantalDierenPASCombi', 0))

        # guard missing AantalDieren
        aantal_dieren = _to_float(_get(ea, 'AantalDieren', 0))
        productie_n = _to_float(_get(ea, 'ProductieN', 0))

        if emissie > 0.0 and aantal_pas_voeding == 0 and aantal_pas_combi == 0 and aantal_dieren > 0:
            if _get(ea, 'Regressie', False):
                # try to find regressieRechte and regressie entries for this diersubcat
                ds_code = (str(_get(ea, 'DierSubCategorieCode', '') or '')).strip()
                regR = _find_first_by_nested_code(regressie_rechte_list, 'TblDierSubCategorie', 'Code', ds_code)
                # fallbacks: if not supplied, try attributes on ea (some PRs put inline values)
                if regR is None:
                    # no regR available -> do no regression reduction
                    regR = None

                if regR is not None:
                    A = _to_float(_get(regR, 'A', None))
                    B = _to_float(_get(regR, 'B', None))
                    # safe compute X
                    X = None
                    if A != 0:
                        X = ((_to_float(productie_n) / aantal_dieren) - B) / A
                    else:
                        X = 0.0

                    # find appropriate TblRegressie entry for this diersubcat
                    reg_candidates = [r for r in regressie_list if str(_get(_get(r, 'TblDierSubCategorie'), 'Code', '')).strip() == ds_code]
                    # choose the first where X < Regessiemax; emulate ordering logic defensively
                    chosen = None
                    try:
                        # try to pick smallest Regessiemax that is >= X
                        reg_candidates_sorted = sorted(reg_candidates, key=lambda r: _to_float(_get(r, 'Regressiemax', 0)))
                        for rc in reg_candidates_sorted:
                            if X is None:
                                continue
                            if X < _to_float(_get(rc, 'Regressiemax', float('inf'))):
                                chosen = rc
                                break
                    except Exception:
                        chosen = None

                    if (
                        chosen is not None
                        and not _starts_with(_get(ea, 'CodeStalType', ''), 'V ')
                        and not _starts_with(_get(ea, 'CodeStalType', ''), 'P ')
                        and _to_float(_get(ea, 'AantalDierenTechniek', 0)) == 0
                    ):
                        reductie = _to_float(_get(chosen, 'Reductiepercentage', 0))
                        emissie = emissie * (1.0 - reductie / 100.0)

            elif _get(ea, 'Convenant', False):
                ds_code = (str(_get(ea, 'DierSubCategorieCode', '') or '')).strip()
                convenant = _find_first_by_nested_code(convenant_list, 'TblDierSubCategorie', 'Code', ds_code)
                if convenant is not None:
                    conv_max = _to_float(_get(convenant, 'Convenantmax', float('inf')))
                    if (productie_n / aantal_dieren) <= conv_max:
                        reductie = _to_float(_get(convenant, 'Reductiepercentage', 0))
                        emissie = emissie * (1.0 - reductie / 100.0)

        # set the computed emission back
        _set(ea, 'EmissieNH3', emissie)
        out.append(ea)

    return out
