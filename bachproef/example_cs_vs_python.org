  for ea in input.EmissieStallen:

    emissie = min(
        ea.TmpEmissieNH3Gereduceerd,
        min(
            ea.TmpEmissieNH3Trad,
            ea.TmpEmissieNH3StalEF
        )
    )

    if (
        emissie > 0.0
        and ea.AantalDierenPASVoeding == 0
        and ea.AantalDierenPASCombi == 0
    ):

        if ea.Regressie:

            regR_matches = [
                r for r in regressieRechteList
                if r.TblDierSubCategorie.Code
                == ea.DierSubCategorieCode
            ]

            if len(regR_matches) != 1:
                raise Exception(
                    "Geen unieke data gevonden voor regressieRechte "
                    "in databank voor diersubcategorie "
                    + ea.DierSubCategorie
                )

            regR = regR_matches[0]

            X = (
                (ea.ProductieN / ea.AantalDieren)
                - regR.B
            ) / regR.A

            regDierSubCatList = [
                r for r in regressieList
                if r.TblDierSubCategorie.Code
                == ea.DierSubCategorieCode
            ]

            mogelijke_regels = [
                r for r in regDierSubCatList
                if X < r.Regressiemax
            ]

            mogelijke_regels.sort(
                key=lambda r: r.Regressiemax
            )

            reg = (
                mogelijke_regels[0]
                if len(mogelijke_regels) > 0
                else None
            )

            if (
                reg is not None
                and not ea.CodeStalType.startswith("V ")
                and not ea.CodeStalType.startswith("P ")
                and ea.AantalDierenTechniek == 0
            ):
                emissie *= (
                    1
                    - reg.Reductiepercentage / 100
                )

        elif ea.Convenant:

            convenant_matches = [
                r for r in convenantList
                if r.TblDierSubCategorie.Code
                == ea.DierSubCategorieCode
            ]

            convenant = (
                convenant_matches[0]
                if len(convenant_matches) > 0
                else None
            )

            if convenant is None:
                raise Exception(
                    "Geen data gevonden voor convenant "
                    "in databank voor diersubcategorie "
                    + ea.DierSubCategorie
                )

            if (
                ea.ProductieN / ea.AantalDieren
                <= convenant.Convenantmax
            ):
                emissie *= (
                    1
                    - convenant.Reductiepercentage / 100
                )

    ea.EmissieNH3 = emissie

return input.EmissieStallen
